Flight::route('POST /admin/catalogo/guardar', function() {
    $auth = Flight::auth();

    }

        $_SESSION['error_msg'] = "Error: El formato de los sinónimos de cantidad no es un JSON válido.";
        Flight::redirect('/admin/catalogo');
        return;
    }

    try {
        $db->beginTransaction();

        // 1. Actualizar las palabras clave de los productos individualmente
        $stmt_update = $db->prepare("UPDATE `productos` SET `palabras_clave` = :keywords WHERE `id` = :id");
        foreach ($keywords as $prod_id => $kws) {
            $stmt_update->execute([
                ':keywords' => trim($kws),
                ':id' => intval($prod_id)
            ]);
        }

        // 2. Obtener la nueva lista consolidada para el snapshot JSON de la versión
        $stmt_snapshot = $db->query("
            SELECT p.`id`, p.`categoria_id`, p.`nombre`, p.`precio`, p.`palabras_clave`, c.`nombre` as categoria_nombre

            JOIN `categorias` c ON p.`categoria_id` = c.`id`

        $delta_hash = md5($json_data . $sinonimos_txt . $umbral_largo . $umbral_corto);

            INSERT INTO `catalogo_versiones`

Flight::route('POST /admin/catalogo/simular', function() {
    $request = Flight::request();
    $data = json_decode($request->getBody(), true) ?? [];

    $frase = trim($data['frase'] ?? '');
    $keywords_map = $data['keywords'] ?? [];
    $sinonimos = $data['sinonimos'] ?? [];
    $umbral_largo = intval($data['umbral_largo'] ?? 3);
    $umbral_corto = intval($data['umbral_corto'] ?? 1);

    if (empty($frase)) {
        Flight::json(['status' => 'error', 'mensaje' => 'La frase está vacía'], 400);
        return;
    }

    // Normalizador de texto
    $normalize = function($str) {
        $str = mb_strtolower($str, 'UTF-8');
        $str = str_replace(
            ['á', 'é', 'í', 'ó', 'ú', 'ü', 'ñ'],
            ['a', 'e', 'i', 'o', 'u', 'u', 'n'],
            $str
        );
        return trim(preg_replace('/\s+/', ' ', $str));
    };

    $normalized_frase = $normalize($frase);

    // 1. Extraer número de Mesa
    $mesa = null;
    if (preg_match('/mesa\s+(\w+|\d+)/i', $normalized_frase, $matches)) {
        $mesa_val = $matches[1];
        if (is_numeric($mesa_val)) {
            $mesa = intval($mesa_val);
        } else {
            $mesa = $sinonimos[$mesa_val] ?? null;
        }
        $normalized_frase = trim(str_replace($matches[0], '', $normalized_frase));
    }

    // 2. Segmentar la orden por conectores "y", "mas", comas
    $segments = preg_split('/\s+y\s+|,\s+|\s+mas\s+/', $normalized_frase);
    $items = [];
    $comanda_json = [];

    // Obtener catálogo base de la BD para mapeo de nombres y precios
    $db = Flight::db();
    $stmt = $db->query("SELECT `id`, `nombre`, `precio` FROM `productos` WHERE `disponible` = 1");
    $db_products = $stmt->fetchAll(PDO::FETCH_ASSOC);

    // Ordenar sinónimos por longitud de texto de mayor a menor
    uksort($sinonimos, function($a, $b) {
        return strlen($b) - strlen($a);
    });

    foreach ($segments as $segment) {
        $segment = trim($segment);
        if (empty($segment)) continue;

        $cantidad = 1;
        $sinonimo_usado = "por defecto (1)";

        // Intentar extraer cantidad por sinónimos de palabras
        $found_qty = false;
        foreach ($sinonimos as $syn => $val) {
            if (strpos($segment, $syn . ' ') === 0) {
                $cantidad = $val;
                $sinonimo_usado = $syn;
                $segment = trim(substr($segment, strlen($syn)));
                $found_qty = true;
                break;
            }
        }

        // Si no se encontró palabra, intentar con dígito directo
        if (!$found_qty) {
            if (preg_match('/^(\d+)\s+/', $segment, $num_matches)) {
                $cantidad = intval($num_matches[1]);
                $sinonimo_usado = $num_matches[1];
                $segment = trim(substr($segment, strlen($num_matches[0])));
            }
        }

        // Extraer notas de tipo "sin cebolla", "con queso"
        $notas = "";
        if (preg_match('/(sin|con)\s+([a-z\s]+)$/i', $segment, $mod_matches)) {
            $notas = $mod_matches[1] . ' ' . trim($mod_matches[2]);
            $segment = trim(str_replace($mod_matches[0], '', $segment));
        }

        // Limpiar conectores en el nombre del producto (de, del, de la)
        $clean_prod_input = trim(preg_replace('/\b(de|del|de\s+la)\b/i', '', $segment));
        $clean_prod_input = preg_replace('/\s+/', ' ', $clean_prod_input);

        if (empty($clean_prod_input)) continue;

        // Búsqueda del mejor candidato Levenshtein
        $best_product = null;
        $min_dist = 999;
        $keywords_coincidente = "";

        foreach ($db_products as $p) {
            $p_id = $p['id'];
            $p_keywords = $keywords_map[$p_id] ?? '';
            $p_keywords_normalized = $normalize($p_keywords);

            if (empty($p_keywords_normalized)) continue;

            // Coincidencia exacta o contención directa
            if (stripos($p_keywords_normalized, $clean_prod_input) !== false) {
                $dist = 0;
                if ($dist < $min_dist) {
                    $min_dist = $dist;
                    $best_product = $p;
                    $keywords_coincidente = $clean_prod_input;
                }
            } else {
                // Levenshtein sobre la cadena de palabras clave completa
                $dist = levenshtein($clean_prod_input, $p_keywords_normalized);
                if ($dist < $min_dist) {
                    $min_dist = $dist;
                    $best_product = $p;
                    $keywords_coincidente = $p_keywords_normalized;
                }

                // Levenshtein a nivel de palabra individual
                $words = explode(' ', $p_keywords_normalized);
                foreach ($words as $w) {
                    $dist = levenshtein($clean_prod_input, $w);
                    if ($dist < $min_dist) {
                        $min_dist = $dist;
                        $best_product = $p;
                        $keywords_coincidente = $w;
                    }
                }
            }
        }

        // Aplicar umbrales configurados
        $matched_product = null;
        $match_valid = false;
        if ($best_product !== null) {
            $len = strlen($clean_prod_input);
            $umbral = ($len <= 5) ? $umbral_corto : $umbral_largo;
            if ($min_dist <= $umbral) {
                $matched_product = $best_product;
                $match_valid = true;
            }
        }

        $items[] = [
            'segmento' => $clean_prod_input,
            'cantidad' => $cantidad,
            'sinonimo_usado' => $sinonimo_usado,
            'notas' => $notas,
            'producto' => $matched_product,
            'distancia' => $min_dist,
            'keywords_coincidente' => $keywords_coincidente
        ];

        if ($match_valid) {
            $comanda_json[] = [
                'producto_id' => intval($matched_product['id']),
                'nombre' => $matched_product['nombre'],
                'cantidad' => $cantidad,
                'precio' => floatval($matched_product['precio']),
                'subtotal' => floatval($matched_product['precio'] * $cantidad),
                'notas' => $notas
            ];
        }
    }

    Flight::json([
        'status' => 'success',
        'frase_original' => $frase,
        'mesa' => $mesa,
        'items' => $items,
        'comanda_json' => $comanda_json
    ]);
});

function procesarTranscripcionFinal(frase) {

        let foundQty = false;

        // Try lookup by synonym words
        for (const syn of sortedSynKeys) {
            if (segment.startsWith(syn + ' ') || segment === syn) {
                cantidad = synonymsCached[syn];
                synonymUsed = syn;
                segment = segment.substring(syn.length).trim();
                foundQty = true;
                break;
            }
        }

        // Try direct digit match

            const numMatch = segment.match(/^(\d+)\s+/);
            if (numMatch) {
                cantidad = parseInt(numMatch[1]);
                synonymUsed = numMatch[1];
                segment = segment.substring(numMatch[0].length).trim();
            }

        // Extract Notes ("sin cebolla", "con queso")

            // Exact match or direct containment
            if (prodKeywords.includes(cleanProductInput)) {
                const dist = 0;
                if (dist < minDist) {
                    minDist = dist;
                    bestProduct = p;
                    matchedKeyword = cleanProductInput;
                }
            } else {
                // Levenshtein on entire keywords
                let dist = levenshtein(cleanProductInput, prodKeywords);
                if (dist < minDist) {
                    minDist = dist;
                    bestProduct = p;
                    matchedKeyword = prodKeywords;
                }

                // Levenshtein on individual words
                const words = prodKeywords.split(' ');
                words.forEach(w => {
                    dist = levenshtein(cleanProductInput, w);
                    if (dist < minDist) {
                        minDist = dist;
                        bestProduct = p;
                        matchedKeyword = w;
                    }
                });
            }
        });

        // Apply configured thresholds
        let matchedProduct = null;
        if (bestProduct) {
            const len = cleanProductInput.length;
            const threshold = (len <= 5) ? umbralCorto : umbralLargo;
            if (minDist <= threshold) {

/**
 * Standard Levenshtein algorithm.
 */
function levenshtein(s1, s2) {
    const track = Array(s2.length + 1).fill(null).map(() => Array(s1.length + 1).fill(null));
    for (let i = 0; i <= s1.length; i++) track[0][i] = i;
    for (let j = 0; j <= s2.length; j++) track[j][0] = j;

    for (let j = 1; j <= s2.length; j++) {
        for (let i = 1; i <= s1.length; i++) {
            const indicator = s1[i - 1] === s2[j - 1] ? 0 : 1;
            track[j][i] = Math.min(
                track[j][i - 1] + 1, // deletion
                track[j - 1][i] + 1, // insertion
                track[j - 1][i - 1] + indicator // substitution
            );
        }
    }
    return track[s2.length][s1.length];
}

        const osc = audioContext.createOscillator();
        const gain = audioContext.createGain();

                segment = segment.substring(syn.length).trim();
                foundQty = true;
                break;
            }
        }

        // Try direct digit match
        if (!foundQty) {
            const numMatch = segment.match(/^(\d+)\s+/);
            if (numMatch) {
                cantidad = parseInt(numMatch[1]);
                synonymUsed = numMatch[1];
                segment = segment.substring(numMatch[0].length).trim();
            }
        }

        // Extract Notes ("sin cebolla", "con queso")
        let notas = "";
        const modMatch = segment.match(/(sin|con)\s+([a-z\s]+)$/i);
        if (modMatch) {
            notas = modMatch[1] + ' ' + modMatch[2].trim();
            segment = segment.replace(modMatch[0], '').trim();
        }

        // Clean connectors
        let cleanProductInput = segment.replace(/\b(de|del|de\s+la)\b/gi, '').replace(/\s+/g, ' ').trim();

        if (cleanProductInput === '') return;

        // Levenshtein Matching
        let bestProduct = null;
        let minDist = 999;
        let matchedKeyword = "";

        catalogCached.forEach(p => {
            const prodKeywords = normalize(p.palabras_clave || '');
            if (!prodKeywords) return;

            // Exact match or direct containment
            if (prodKeywords.includes(cleanProductInput)) {
                const dist = 0;
                if (dist < minDist) {
                    minDist = dist;
                    bestProduct = p;
                    matchedKeyword = cleanProductInput;
                }
            } else {
                // Levenshtein on entire keywords
                let dist = levenshtein(cleanProductInput, prodKeywords);
                if (dist < minDist) {
                    minDist = dist;
                    bestProduct = p;
                    matchedKeyword = prodKeywords;
                }

                // Levenshtein on individual words
                const words = prodKeywords.split(' ');
                words.forEach(w => {
                    dist = levenshtein(cleanProductInput, w);
                    if (dist < minDist) {
                        minDist = dist;
                        bestProduct = p;
                        matchedKeyword = w;
                    }
                });
            }
        });

        // Apply configured thresholds
        let matchedProduct = null;
        if (bestProduct) {
            const len = cleanProductInput.length;
            const threshold = (len <= 5) ? umbralCorto : umbralLargo;
            if (minDist <= threshold) {
                matchedProduct = bestProduct;
            }
        }

        if (matchedProduct) {
            orderItems.push({
                producto_id: matchedProduct.id,
                nombre: matchedProduct.nombre,
                cantidad: cantidad,
                precio: parseFloat(matchedProduct.precio),
                subtotal: parseFloat(matchedProduct.precio * cantidad),
                notas: notas
            });
        }
    });

    if (orderItems.length > 0) {
        renderComandaPrevia(mesa, orderItems);
    }
}

/**
 * Standard Levenshtein algorithm.
 */
function levenshtein(s1, s2) {
    const track = Array(s2.length + 1).fill(null).map(() => Array(s1.length + 1).fill(null));
    for (let i = 0; i <= s1.length; i++) track[0][i] = i;
    for (let j = 0; j <= s2.length; j++) track[j][0] = j;

    for (let j = 1; j <= s2.length; j++) {
        for (let i = 1; i <= s1.length; i++) {
            const indicator = s1[i - 1] === s2[j - 1] ? 0 : 1;
            track[j][i] = Math.min(
                track[j][i - 1] + 1, // deletion
                track[j - 1][i] + 1, // insertion
                track[j - 1][i - 1] + indicator // substitution
            );
        }
    }
    return track[s2.length][s1.length];
}
